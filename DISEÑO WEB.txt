/** Escapa texto dinámico para evitar XSS */
function esc(str) {
  if (str === null || str === undefined) return "";
  return String(str)
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&#39;");
}

/** Formatea un número con separadores de miles */
function fmt(n) {
  const num = Number(n) || 0;
  return num.toLocaleString("es-ES", { maximumFractionDigits: 0 });
}

/** Formatea una fecha "yyyy-mm-dd" a texto legible */
function fmtDate(str) {
  if (!str) return "";
  const [y, m, d] = str.split("-").map(Number);
  const date = new Date(y, m - 1, d);
  return date.toLocaleDateString("es-ES", {
    weekday: "long",
    day: "numeric",
    month: "long",
    year: "numeric"
  });
}

function fmtDateShort(str) {
  if (!str) return "";
  const [y, m, d] = str.split("-").map(Number);
  const date = new Date(y, m - 1, d);
  return date.toLocaleDateString("es-ES", { day: "2-digit", month: "short" });
}

/** Hoy en formato yyyy-mm-dd (local, sin UTC) */
function todayISO() {
  const d = new Date();
  const y = d.getFullYear();
  const m = String(d.getMonth() + 1).padStart(2, "0");
  const day = String(d.getDate()).padStart(2, "0");
  return `${y}-${m}-${day}`;
}

/** Suma días a una fecha ISO */
function addDays(iso, days) {
  const [y, m, d] = iso.split("-").map(Number);
  const date = new Date(y, m - 1, d);
  date.setDate(date.getDate() + days);
  const yy = date.getFullYear();
  const mm = String(date.getMonth() + 1).padStart(2, "0");
  const dd = String(date.getDate()).padStart(2, "0");
  return `${yy}-${mm}-${dd}`;
}

/** Emoji por categoría (heurística por palabras clave) */
function catEmoji(name) {

  if (/redes|social/.test(n)) return "📵";
  if (/ortograf|ortogr/.test(n)) return "✍️";
  if (/gym|gimnasio|entreno/.test(n)) return "💪";
  if (/ausen/.test(n)) return "😴";

  if (/tardanza|tarde|llegar/.test(n)) return "⏰";

}

/* ==========================================================================
   CÁLCULOS
   ========================================================================== */
function computeStats(person) {
  const records = person.records || [];
  let total = 0, won = 0, lost = 0;
  const byDay = {}; // { "yyyy-mm-dd": total }
  const byCat = {}; // { nombre: { count, points, name } }

  records.forEach(r => {
    total += Number(r.points) || 0;
    if (r.points > 0) won += Number(r.points);
    if (r.points < 0) lost += Math.abs(Number(r.points));
    byDay[r.date] = (byDay[r.date] || 0) + (Number(r.points) || 0);
    if (!byCat[r.category]) {
      byCat[r.category] = { name: r.category, count: 0, points: 0 };
    }
    byCat[r.category].count += 1;
    byCat[r.category].points += Number(r.points) || 0;
  });

  // Racha: días consecutivos hacia atrás (incluyendo hoy) donde hubo registro positivo
  // Se cuentan días con al menos 1 punto positivo neto.
  let streak = 0;
  let d = todayISO();
  while (true) {
    const dayTotal = byDay[d];
    if (dayTotal && dayTotal > 0) {
      streak += 1;
      d = addDays(d, -1);
    } else if (dayTotal === 0 || dayTotal === undefined) {
      // Si hoy todavía no hay registros, permitimos empezar desde ayer
      if (streak === 0 && d === todayISO()) {
        d = addDays(d, -1);
        continue;
      }
      break;
    } else {
      // Día con saldo negativo rompe la racha
      break;
    }
  }

  // Mejor día: el de mayor saldo neto positivo
  let bestDay = { date: null, points: 0 };
  Object.entries(byDay).forEach(([date, pts]) => {
    if (pts > bestDay.points) bestDay = { date, points: pts };
  });

  // Porcentaje respecto a la meta
  const goal = Number(person.goal) || 2000;
  const pct = goal > 0 ? Math.min(100, Math.max(0, (total / goal) * 100)) : 0;

  // Nivel
  let level = { emoji: "🌱", name: "Comenzando" };
  if (total >= goal) level = { emoji: "🏆", name: "Meta alcanzada" };
  else if (pct >= 75) level = { emoji: "🏁", name: "Recta final" };
  else if (pct >= 50) level = { emoji: "⚡", name: "A mitad de camino" };
  else if (pct >= 25) level = { emoji: "🔥", name: "En marcha" };

  // Últimos 30 días (serie acumulada)
  const todayDate = state.today;
  const labels = [];
  const accum = [];
  const daily = [];
  let running = 0;

  // Suma de registros anteriores a hace 30 días
  const cutoff = addDays(todayDate, -29);
  records.forEach(r => {
    if (r.date < cutoff) running += Number(r.points) || 0;
  });

  for (let i = -29; i <= 0; i++) {
    const day = addDays(todayDate, i);
    labels.push(day);
    running += (byDay[day] || 0);
    accum.push(running);
    daily.push(byDay[day] || 0);
  }

  const todayCount = (records.filter(r => r.date === todayDate)).length;

  return {
    total, won, lost,
    today: { points: byDay[todayDate] || 0, count: todayCount },
    streak, bestDay,
    goal, pct, level,
    byCat, byDay,
    series: { labels, accum, daily }
  };
}

function toast(message, type = "info", duration = 3500) {
  const container = $("#toasts");
  const el = document.createElement("div");
  el.className = `toast ${type}`;
  const icons = { error: "⚠️", success: "✅", info: "ℹ️" };
  el.innerHTML = `
    <span class="toast-icon">${icons[type] || icons.info}</span>
    <div class="toast-msg">${esc(message)}</div>
  `;
  container.appendChild(el);
  setTimeout(() => {
    el.classList.add("removing");
    setTimeout(() => el.remove(), 250);
  }, duration);
}

const overlay = $("#overlay");
const modal = $("#modal");
let lastFocus = null;
let modalCleanup = null;

function openModal(html, opts = {}) {
  lastFocus = document.activeElement;
  modal.innerHTML = html;
  overlay.classList.toggle("bottom-sheet", !!opts.bottomSheet);
  overlay.classList.add("open");
  document.body.style.overflow = "hidden";

  // Focus al primer input
  requestAnimationFrame(() => {
    const first = modal.querySelector("input, textarea, select, button");
    if (first) first.focus();
  });

  if (opts.onMount) opts.onMount(modal);
  modalCleanup = opts.onClose || null;
}

function closeModal() {
  overlay.classList.remove("open");
  document.body.style.overflow = "";
  if (modalCleanup) { try { modalCleanup(); } catch(e) {} modalCleanup = null; }
  if (lastFocus) { try { lastFocus.focus(); } catch(e) {} }
  setTimeout(() => { if (!overlay.classList.contains("open")) modal.innerHTML = ""; }, 300);
}

overlay.addEventListener("click", (e) => {
  if (e.target === overlay) closeModal();
});
document.addEventListener("keydown", (e) => {
  if (e.key === "Escape" && overlay.classList.contains("open")) closeModal();
});

/** Añade un registro */
function apiAddRecord(key, payload) {
  return new Promise((resolve, reject) => {
    if (MOCK.enabled) {
      setTimeout(() => {
        const person = MOCK.data.persons.find(p => p.key === key);
        if (!person) return reject(new Error("Persona no encontrada"));
        // Validar grupo único por día
        const cat = person.categories.find(c => c.name === payload.category);
        if (cat && cat.group) {
          const exists = person.records.some(r =>
            r.date === payload.date &&
            person.categories.find(c => c.name === r.category)?.group === cat.group
          );
          if (exists) {
            return reject(new Error(`Ya registraste esta categoría (${cat.group}) en esa fecha`));
          }
        }
        const newId = person.records.length ? Math.max(...person.records.map(r => r.id)) + 1 : 1;
        const record = {
          id: newId,
          date: payload.date,
          category: payload.category,
          points: payload.points === "" ? (cat.points || 0) : Number(payload.points),
          reason: payload.reason || ""
        };
        person.records.push(record);
        resolve(structuredClone(person));
      }, 250);
      return;
    }
    try {
      google.script.run
        .withSuccessHandler(resolve)
        .withFailureHandler((e) => reject(new Error(e && e.message ? e.message : "Error al guardar")))
        .addRecord(key, payload);
    } catch (e) {
      reject(new Error("No se pudo conectar con el backend"));
    }
  });
}

   ACCIONES
   ========================================================================== */

async function submitRecord({ date, category, points, reason }) {
  const person = currentPerson();
  const cat = person.categories.find(c => c.name === category);
  const dateVal = date || $("#reg-date").value;
  if (!dateVal) { toast("Selecciona una fecha", "error"); return; }

  // Validación de grupo (cliente)
  if (cat && cat.group) {
    const exists = person.records.some(r =>
      r.date === dateVal &&
      person.categories.find(c => c.name === r.category)?.group === cat.group
    );
    if (exists) {
      toast(`Ya registraste esta categoría el ${fmtDateShort(dateVal)}`, "error");
      return;
    }
  }

  // Si la categoría no tiene puntos fijos y no se pasó points, usar el input o el default
  let ptsVal;
  if (cat.points === null) {
    ptsVal = points;
  } else {
    ptsVal = points !== undefined ? points : cat.points;
  }

  try {
    const updated = await apiAddRecord(person.key, {
      date: dateVal,
      category,
      points: ptsVal === "" ? "" : Number(ptsVal),
      reason: reason || ""
    });
    // Reemplazar la persona en el estado
    const idx = state.persons.findIndex(p => p.key === person.key);
    if (idx >= 0) state.persons[idx] = updated;
    toast(`Registro añadido (+${ptsVal >= 0 ? "" : ""}${fmt(ptsVal)})`, "success");
    render();
  } catch (e) {
    toast(e.message || "No se pudo guardar", "error");
  }
}

async function deleteRecordById(id) {
  const person = currentPerson();
  try {
    const updated = await apiDeleteRecord(person.key, id);
    const idx = state.persons.findIndex(p => p.key === person.key);
    if (idx >= 0) state.persons[idx] = updated;
    toast("Registro eliminado", "success");
    render();
  } catch (e) {
    toast(e.message || "No se pudo eliminar", "error");
  }
}

      <div class="empty">
        <div class="emoji">📊</div>
        <h3>Sin datos todavía</h3>
        <p>Empieza a registrar para ver el desglose por categoría.</p>
      </div>`;
    return;
  }
  // Ordenar por número de registros
  cats.sort((a, b) => b.count - a.count);
  const maxCount = Math.max(...cats.map(c => c.count));
  container.innerHTML = cats.map(c => `
    <div class="cat-row">
      <span class="emoji" aria-hidden="true">${catEmoji(c.name)}</span>
      <div class="info">
        <div class="name">${esc(c.name)}</div>
        <div class="bar"><div class="bar-fill" style="width: ${(c.count / maxCount) * 100}%"></div></div>
        <div class="count">${fmt(c.count)} ${c.count === 1 ? "registro" : "registros"}</div>
      </div>
      <div class="points num ${c.points >= 0 ? "pos" : "neg"}">${c.points >= 0 ? "+" : ""}${fmt(c.points)}</div>
    </div>
  `).join("");
}

function renderHistory(person) {
  const container = $("#history-list");
  const recs = [...(person.records || [])].sort((a, b) => {
    if (a.date !== b.date) return b.date.localeCompare(a.date);
    return b.id - a.id;
  }).slice(0, 30);
  if (recs.length === 0) {
    container.innerHTML = `
      <div class="empty">
        <div class="emoji">🗒️</div>
        <h3>Aún no hay registros</h3>
        <p>Registra tu primer punto desde la sección de arriba.</p>
      </div>`;
    return;
  }
  container.innerHTML = recs.map(r => `
    <article class="history-item" data-id="${r.id}">
      <span class="emoji" aria-hidden="true">${catEmoji(r.category)}</span>
      <div class="info">
        <div class="cat-name">${esc(r.category)}</div>
        ${r.reason ? `<div class="reason">${esc(r.reason)}</div>` : ""}
        <div class="meta">
          <span>${esc(fmtDateShort(r.date))}</span>
        </div>
      </div>
      <div style="display:flex; align-items:center; gap:6px;">
        <div class="pts num ${r.points >= 0 ? "pos" : "neg"}">${r.points >= 0 ? "+" : ""}${fmt(r.points)}</div>
        <button type="button" class="btn-delete" data-id="${r.id}" aria-label="Eliminar registro de ${esc(r.category)}">
          <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><polyline points="3 6 5 6 21 6"/><path d="M19 6l-2 14a2 2 0 0 1-2 2H9a2 2 0 0 1-2-2L5 6"/><path d="M10 11v6M14 11v6"/><path d="M9 6V4a2 2 0 0 1 2-2h2a2 2 0 0 1 2 2v2"/></svg>
        </button>
      </div>
    </article>
  `).join("");

  $$(".btn-delete", container).forEach(btn => {
    btn.addEventListener("click", () => {
      const id = Number(btn.getAttribute("data-id"));
      const rec = person.records.find(r => r.id === id);
      if (rec) openDeleteModal(rec);
    });
  });
}

/* ==========================================================================
   GRÁFICOS (Chart.js)
   ========================================================================== */
let chartLine = null, chartBars = null;

function getChartColors() {
  const isDark = window.matchMedia("(prefers-color-scheme: dark)").matches;
  return {
    text: isDark ? "#b8bfbb" : "#4a5451",
    grid: isDark ? "rgba(255,255,255,0.08)" : "rgba(0,0,0,0.06)",
    pos: "#2f8f4f",
    neg: "#c7523a",
    forest: isDark ? "#3fa076" : "#1f5d3d",
    gold: isDark ? "#e0b868" : "#c9a24f"
  };
}

function renderCharts(stats) {
  if (typeof Chart === "undefined") return;
  const colors = getChartColors();
  const labels = stats.series.labels.map(fmtDateShort);

  // Destruir instancias previas para evitar memory leaks
  if (chartLine) { chartLine.destroy(); chartLine = null; }
  if (chartBars) { chartBars.destroy(); chartBars = null; }

  const commonOpts = {
    responsive: true,
    maintainAspectRatio: false,
    plugins: {
      legend: { display: false },
      tooltip: {
        backgroundColor: "rgba(23, 31, 27, 0.95)",
        titleColor: "#fff",
        bodyColor: "#fff",
        borderColor: "rgba(255,255,255,0.1)",
        borderWidth: 1,
        padding: 10,
        cornerRadius: 8,
        titleFont: { family: "Manrope, system-ui", weight: "600" },
        bodyFont: { family: "Manrope, system-ui" },
        callbacks: {
          label: (ctx) => {
            const val = ctx.parsed.y;
            return `${ctx.dataset.label || ""}: ${val >= 0 ? "+" : ""}${fmt(val)} pts`;
          }
        }
      }
    },
    scales: {
      x: {
        ticks: { color: colors.text, font: { family: "Manrope, system-ui", size: 10 }, maxRotation: 0, autoSkipPadding: 12 },
        grid: { display: false },
        border: { display: false }
      },
      y: {
        ticks: { color: colors.text, font: { family: "Manrope, system-ui", size: 10 }, callback: (v) => fmt(v) },
        grid: { color: colors.grid },
        border: { display: false }
      }
    }
  };

  // Línea acumulada
  const lineCtx = $("#chart-line").getContext("2d");
  const gradient = lineCtx.createLinearGradient(0, 0, 0, 240);
  gradient.addColorStop(0, colors.forest + "55");
  gradient.addColorStop(1, colors.forest + "00");
  chartLine = new Chart(lineCtx, {
    type: "line",
    data: {
      labels,
      datasets: [{
        label: "Acumulado",
        data: stats.series.accum,
        borderColor: colors.forest,
        backgroundColor: gradient,
        borderWidth: 2.5,
        fill: true,
        tension: 0.35,
        pointRadius: 0,
        pointHitRadius: 10
      }]
    },
    options: { ...commonOpts }
  });

  // Barras por día
  const barColors = stats.series.daily.map(v => v >= 0 ? colors.pos : colors.neg);
  chartBars = new Chart($("#chart-bars").getContext("2d"), {
    type: "bar",
    data: {
      labels,
      datasets: [{
        label: "Puntos",
        data: stats.series.daily,
        backgroundColor: barColors,
        borderRadius: 4,
        borderSkipped: false,
        maxBarThickness: 18
      }]
    },
    options: {
      ...commonOpts,
      plugins: {
        ...commonOpts.plugins,
        tooltip: {
          ...commonOpts.plugins.tooltip,
          callbacks: {
            label: (ctx) => {
              const val = ctx.parsed.y;
              return `${val >= 0 ? "+" : ""}${fmt(val)} pts`;
            }
          }
        }
      }
    }
  });
}

/* ==========================================================================
   CONFETI
   ========================================================================== */
function triggerConfetti() {
  const canvas = $("#confetti");
  const ctx = canvas.getContext("2d");
  canvas.width = window.innerWidth;
  canvas.height = window.innerHeight;
  const particles = [];
  const colors = ["#c9a24f", "#e0c07a", "#1f5d3d", "#2d7a52", "#fff5d1"];
  for (let i = 0; i < 120; i++) {
    particles.push({
      x: canvas.width / 2 + (Math.random() - 0.5) * 200,
      y: canvas.height / 2,
      vx: (Math.random() - 0.5) * 12,
      vy: -Math.random() * 14 - 6,
      g: 0.35,
      size: Math.random() * 6 + 3,
      color: colors[Math.floor(Math.random() * colors.length)],
      rot: Math.random() * Math.PI * 2,
      vr: (Math.random() - 0.5) * 0.3,
      life: 1
    });
  }
  let frame;
  const start = performance.now();
  function step(now) {
    const elapsed = now - start;
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    particles.forEach(p => {
      p.x += p.vx;
      p.y += p.vy;
      p.vy += p.g;
      p.rot += p.vr;
      p.life = Math.max(0, 1 - elapsed / 3500);
      ctx.save();
      ctx.translate(p.x, p.y);
      ctx.rotate(p.rot);
      ctx.globalAlpha = p.life;
      ctx.fillStyle = p.color;
      ctx.fillRect(-p.size / 2, -p.size / 2, p.size, p.size * 0.6);
      ctx.restore();
    });
    if (elapsed < 4000) {
      frame = requestAnimationFrame(step);
    } else {
      ctx.clearRect(0, 0, canvas.width, canvas.height);
    }
  }
  frame = requestAnimationFrame(step);
  // Respetar prefers-reduced-motion
  if (window.matchMedia("(prefers-reduced-motion: reduce)").matches) {
    cancelAnimationFrame(frame);
    ctx.clearRect(0, 0, canvas.width, canvas.height);
  }
}

function renderPersonSelector() {
  const container = $(".person-selector");
  const pill = $("#selector-pill");
  // Eliminar botones previos
  $$(".person-btn", container).forEach(b => b.remove());

  state.persons.forEach(p => {
    const btn = document.createElement("button");
    btn.type = "button";
    btn.className = "person-btn";
    btn.setAttribute("role", "tab");
    btn.setAttribute("aria-selected", p.key === state.selectedKey ? "true" : "false");
    btn.setAttribute("data-key", p.key);
    btn.innerHTML = `
      <span class="avatar">${esc(p.name.charAt(0).toUpperCase())}</span>
      <span class="person-name">${esc(p.name)}</span>
    `;
    btn.addEventListener("click", () => {
      state.selectedKey = p.key;
      renderPersonSelector();
      render();
    });
    container.appendChild(btn);
  });

  // Posicionar pill
  requestAnimationFrame(() => {
    const active = container.querySelector('.person-btn[aria-selected="true"]');
    if (active) {
      pill.style.width = active.offsetWidth + "px";
      pill.style.transform = `translateX(${active.offsetLeft - 4}px)`;
    }
  });
}

/* ==========================================================================
   INICIALIZACIÓN
   ========================================================================== */
async function init() {
  // Header scroll detection
  const header = $("#header");
  const onScroll = () => {
    if (window.scrollY > 10) header.classList.add("scrolled");
    else header.classList.remove("scrolled");
  };
  window.addEventListener("scroll", onScroll, { passive: true });
  onScroll();

  // Fecha por defecto
  $("#reg-date").value = state.today;
  $("#reg-date").max = state.today; // no permitir fechas futuras por defecto (pero se puede cambiar en manual)

  // Botón registro manual
  $("#btn-manual").addEventListener("click", openManualModal);

  // Resize para redibujar charts
  let resizeT;
  window.addEventListener("resize", () => {
    clearTimeout(resizeT);
    resizeT = setTimeout(() => {
      if (chartLine) chartLine.resize();
      if (chartBars) chartBars.resize();
      // Redimensionar confetti canvas
      const cv = $("#confetti");
      if (cv) { cv.width = window.innerWidth; cv.height = window.innerHeight; }
    }, 150);
  });

  // Cargar datos
  try {
    const data = await apiGetAll();
    state.today = data.today || todayISO();
    state.persons = data.persons || [];
    state.selectedKey = state.persons[0]?.key || null;
    state.loaded = true;

    // Ocultar loader, mostrar dashboard
    $("#initial-loading").classList.add("hidden");
    $("#dashboard").classList.remove("hidden");

    // Renderizar selector y dashboard
    renderPersonSelector();
    render();

    if (MOCK.enabled) {
      toast("Modo demo activo. Los cambios no se guardan.", "info", 4000);
    }
  } catch (e) {
    $("#initial-loading").innerHTML = `
      <div class="empty" style="padding: 24px;">
        <div class="emoji">⚠️</div>
        <h3>Error al cargar</h3>
        <p>${esc(e.message || "No se pudieron obtener los datos.")}</p>
      </div>
    `;
    toast(e.message || "Error al cargar", "error", 5000);
  }
}

// Cambio de tema dinámico (re-renderizar charts con nuevos colores)

  if (state.loaded) {
    const person = currentPerson();
    if (person) renderCharts(computeStats(person));
  }
});

// Arranque
document.addEventListener("DOMContentLoaded", init);
