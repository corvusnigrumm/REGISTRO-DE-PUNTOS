function esc(str) {
  if (str === null || str === undefined) return "";
  return String(str)
    .replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;").replace(/'/g, "&#39;");
}
function fmt(n) {
  const num = Number(n) || 0;
  return num.toLocaleString("es-ES", { maximumFractionDigits: 0 });
}
function fmtDate(str) {
  if (!str) return "";
  const [y, m, d] = str.split("-").map(Number);
  return new Date(y, m - 1, d).toLocaleDateString("es-ES", {
    weekday: "long", day: "numeric", month: "long", year: "numeric"
  });
}
function fmtDateShort(str) {
  if (!str) return "";
  const [y, m, d] = str.split("-").map(Number);
  return new Date(y, m - 1, d).toLocaleDateString("es-ES", { day: "2-digit", month: "short" });
}
function todayISO() {
  const d = new Date();
  return `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,"0")}-${String(d.getDate()).padStart(2,"0")}`;
}
function addDays(iso, days) {
  const [y, m, d] = iso.split("-").map(Number);
  const date = new Date(y, m - 1, d);
  date.setDate(date.getDate() + days);
  return `${date.getFullYear()}-${String(date.getMonth()+1).padStart(2,"0")}-${String(date.getDate()).padStart(2,"0")}`;
}
function catEmoji(name) {
  const n = (name || "").toLowerCase();
  if (/redes|social/.test(n)) return "📵";

/** Animación count-up de un número */
function animateNumber(el, from, to, duration = 900, formatter = fmt) {
  if (window.matchMedia("(prefers-reduced-motion: reduce)").matches) {
    el.textContent = formatter(to);
    return;
  }
  const start = performance.now();
  const diff = to - from;
  function step(now) {
    const t = Math.min(1, (now - start) / duration);
    const eased = 1 - Math.pow(1 - t, 3);
    el.textContent = formatter(Math.round(from + diff * eased));
    if (t < 1) requestAnimationFrame(step);
  }
  requestAnimationFrame(step);
}

/** Efecto "+X" flotante desde un elemento */
function floatPoints(sourceEl, points) {
  if (window.matchMedia("(prefers-reduced-motion: reduce)").matches) return;
  const rect = sourceEl.getBoundingClientRect();
  const el = document.createElement("div");
  el.className = `float-points ${points >= 0 ? "pos" : "neg"}`;
  el.textContent = `${points >= 0 ? "+" : ""}${fmt(points)}`;
  el.style.left = (rect.left + rect.width / 2 - 30) + "px";
  el.style.top = (rect.top + rect.height / 2 - 16) + "px";
  document.body.appendChild(el);
  setTimeout(() => el.remove(), 1200);
}

/** Efecto ripple en un botón */
function addRipple(btn, e) {
  if (window.matchMedia("(prefers-reduced-motion: reduce)").matches) return;
  const rect = btn.getBoundingClientRect();
  const size = Math.max(rect.width, rect.height);
  const x = (e.clientX || rect.left + rect.width/2) - rect.left - size/2;

  r.style.left = x + "px";

  btn.appendChild(r);
  setTimeout(() => r.remove(), 700);
}

function computeStats(person) {
  const records = person.records || [];
  let total = 0, won = 0, lost = 0;
  const byDay = {};
  const byCat = {};

  records.forEach(r => {
    total += Number(r.points) || 0;
    if (r.points > 0) won += Number(r.points);
    if (r.points < 0) lost += Math.abs(Number(r.points));
    byDay[r.date] = (byDay[r.date] || 0) + (Number(r.points) || 0);
    if (!byCat[r.category]) byCat[r.category] = { name: r.category, count: 0, points: 0 };
    byCat[r.category].count += 1;
    byCat[r.category].points += Number(r.points) || 0;
  });

  let streak = 0;
  let d = todayISO();
  while (true) {
    const dayTotal = byDay[d];
    if (dayTotal && dayTotal > 0) {
      streak += 1;
      d = addDays(d, -1);
    } else if ((dayTotal === 0 || dayTotal === undefined) && streak === 0 && d === todayISO()) {
      d = addDays(d, -1);
      continue;
    } else {
      break;
    }
  }

  let bestDay = { date: null, points: 0 };
  Object.entries(byDay).forEach(([date, pts]) => {
    if (pts > bestDay.points) bestDay = { date, points: pts };
  });

  const goal = Number(person.goal) || 2000;
  const pct = goal > 0 ? Math.min(100, Math.max(0, (total / goal) * 100)) : 0;

  let level = { emoji: "🌱", name: "Comenzando" };
  if (total >= goal) level = { emoji: "🏆", name: "Meta alcanzada" };
  else if (pct >= 75) level = { emoji: "🏁", name: "Recta final" };
  else if (pct >= 50) level = { emoji: "⚡", name: "A mitad de camino" };
  else if (pct >= 25) level = { emoji: "🔥", name: "En marcha" };

  const todayDate = state.today;
  const labels = [], accum = [], daily = [];
  let running = 0;
  const cutoff = addDays(todayDate, -29);
  records.forEach(r => { if (r.date < cutoff) running += Number(r.points) || 0; });
  for (let i = -29; i <= 0; i++) {
    const day = addDays(todayDate, i);
    labels.push(day);
    running += (byDay[day] || 0);
    accum.push(running);
    daily.push(byDay[day] || 0);
  }

  const todayCount = records.filter(r => r.date === todayDate).length;

  return {
    total, won, lost,
    today: { points: byDay[todayDate] || 0, count: todayCount },
    streak, bestDay, goal, pct, level, byCat, byDay,
    series: { labels, accum, daily }
  };
}

      <button type="button" class="btn btn-primary" id="man-save">Guardar</button>
    </div>
  `;
  openModal(html, {
    bottomSheet: true,
    onMount: (m) => {
      const catSel = m.querySelector("#man-cat");
      const pointsWrap = m.querySelector("#man-points-wrap");
      const pointsInput = m.querySelector("#man-points");
      const updatePtsField = () => {
        const cat = person.categories.find(c => c.name === catSel.value);
        if (cat && cat.points === null) {
          pointsWrap.style.display = "block";
        } else {
          pointsWrap.style.display = "none";
          pointsInput.value = "";
        }
      };
      catSel.addEventListener("change", updatePtsField);
      updatePtsField();
      m.querySelector("#man-cancel").addEventListener("click", closeModal);
      m.querySelector("#modal-close-x").addEventListener("click", closeModal);
      m.querySelector("#man-save").addEventListener("click", async () => {
        const catName = catSel.value;
        const cat = person.categories.find(c => c.name === catName);
        const date = m.querySelector("#man-date").value;
        const reason = m.querySelector("#man-reason").value.trim();
        if (!date) { toast("Selecciona una fecha", "error"); return; }
        let points;
        if (cat.points === null) {
          const v = m.querySelector("#man-points").value.trim();
          if (v === "" || isNaN(Number(v))) { toast("Ingresa los puntos", "error"); return; }
          points = Number(v);
        } else {
          points = cat.points;
        }
        await submitRecord({ date, category: catName, points, reason }, null);
        closeModal();
      });
    }
  });
}

function buildMockData() {
  const today = todayISO();
  const commonCats = [
    { name: "Día sin redes sociales", points: 50, group: "sinRedes" },
    { name: "Día de gym", points: 20, group: "gym" },
    { name: "Buena acción", points: 30, group: null },
    { name: "Error ortográfico", points: -15, group: null },
    { name: "Ausencia", points: -50, group: null },
    { name: "Otro (a criterio)", points: null, group: null }
  ];
  function randRecords(seed) {
    const recs = [];
    let id = 0;
    for (let i = -28; i <= 0; i++) {
      const date = addDays(today, i);
      const r = Math.abs(Math.sin(seed + i * 0.7));
      if (r < 0.6) {
        const cat = commonCats[Math.floor(Math.abs(Math.sin(seed + i)) * (commonCats.length - 1))];
        const reason = ["", "Por la mañana", "Antes de dormir", "Reto del día", "Se esforzó", "Sin excusas"][Math.floor(Math.abs(Math.sin(seed + i * 2.3)) * 6)];
        recs.push({
          id: id++, date, category: cat.name,
          points: cat.points === null ? (Math.floor(r * 40) - 10) : cat.points,
          reason
        });
      }
    }
    return recs;
  }
  MOCK.data = {
    today,
    persons: [
      { key: "p1", name: "Leonardo", goal: 2000, reward: "Un juego con valor máximo de 200K", categories: commonCats, records: randRecords(13) },
      { key: "p2", name: "Felipe", goal: 2000, reward: "Cena gourmet en restaurante a elección", categories: commonCats, records: randRecords(7) }
    ]
  };
}

function apiGetAll() {
  return new Promise((resolve, reject) => {
    if (MOCK.enabled) {
      if (!MOCK.data) buildMockData();
      setTimeout(() => resolve(structuredClone(MOCK.data)), 600);
      return;
    }
    try {
      google.script.run
        .withSuccessHandler(resolve)
        .withFailureHandler((e) => reject(new Error(e && e.message ? e.message : "Error desconocido")))
        .getAll();
    } catch (e) { reject(new Error("No se pudo conectar con el backend")); }
  });
}

function apiAddRecord(key, payload) {
  return new Promise((resolve, reject) => {
    if (MOCK.enabled) {
      setTimeout(() => {
        const person = MOCK.data.persons.find(p => p.key === key);
        if (!person) return reject(new Error("Persona no encontrada"));
        const cat = person.categories.find(c => c.name === payload.category);
        if (cat && cat.group) {
          const exists = person.records.some(r =>
            r.date === payload.date &&
            (person.categories.find(c => c.name === r.category) || {}).group === cat.group
          );
          if (exists) return reject(new Error(`Ya registraste "${cat.name}" en esa fecha (grupo ${cat.group})`));
        }
        const newId = person.records.length ? Math.max(...person.records.map(r => r.id)) + 1 : 1;
        person.records.push({
          id: newId,
          date: payload.date,
          category: payload.category,
          points: payload.points === "" ? (cat.points || 0) : Number(payload.points),
          reason: payload.reason || ""
        });
        resolve(structuredClone(person));
      }, 300);
      return;
    }
    try {
      google.script.run
        .withSuccessHandler(resolve)
        .withFailureHandler((e) => reject(new Error(e && e.message ? e.message : "Error al guardar")))
        .addRecord(key, payload);
    } catch (e) { reject(new Error("No se pudo conectar con el backend")); }
  });
}

function apiDeleteRecord(key, id) {
  return new Promise((resolve, reject) => {
    if (MOCK.enabled) {
      setTimeout(() => {
        const person = MOCK.data.persons.find(p => p.key === key);
        if (!person) return reject(new Error("Persona no encontrada"));
        person.records = person.records.filter(r => r.id !== id);
        resolve(structuredClone(person));
      }, 250);
      return;
    }
    try {
      google.script.run
        .withSuccessHandler(resolve)
        .withFailureHandler((e) => reject(new Error(e && e.message ? e.message : "Error al eliminar")))
        .deleteRecord(key, id);
    } catch (e) { reject(new Error("No se pudo conectar con el backend")); }
  });
}

async function submitRecord({ date, category, points, reason }, sourceEl) {
  const person = currentPerson();
  const cat = person.categories.find(c => c.name === category);
  const dateVal = date || $("#reg-date").value;
  if (!dateVal) { toast("Selecciona una fecha", "error"); return; }

  if (cat && cat.group) {
    const exists = person.records.some(r =>
      r.date === dateVal &&
      (person.categories.find(c => c.name === r.category) || {}).group === cat.group
    );
    if (exists) {
      toast(`Ya registraste esta categoría el ${fmtDateShort(dateVal)}`, "error");
      return;
    }
  }

  let ptsVal;
  if (cat.points === null) ptsVal = points;
  else ptsVal = points !== undefined ? points : cat.points;

  try {
    const updated = await apiAddRecord(person.key, {
      date: dateVal, category,
      points: ptsVal === "" ? "" : Number(ptsVal),
      reason: reason || ""
    });
    const idx = state.persons.findIndex(p => p.key === person.key);
    if (idx >= 0) state.persons[idx] = updated;

    // Animación de puntos flotantes desde el botón pulsado
    const numPts = Number(ptsVal);
    if (sourceEl) {
      floatPoints(sourceEl, numPts);
    } else {
      // Si viene de modal, flotar desde el anillo
      floatPoints($(".ring-wrap"), numPts);
    }
    toast(`Registrado: ${numPts >= 0 ? "+" : ""}${fmt(numPts)} pts`, "success");
    $("#reg-note").value = "";
    render();
  } catch (e) {
    toast(e.message || "No se pudo guardar", "error");
  }
}

async function deleteRecordById(id) {
  const person = currentPerson();
  try {
    const updated = await apiDeleteRecord(person.key, id);
    const idx = state.persons.findIndex(p => p.key === person.key);
    if (idx >= 0) state.persons[idx] = updated;
    toast("Registro eliminado", "success");
    render();
  } catch (e) {
    toast(e.message || "No se pudo eliminar", "error");
  }
}

let prevPct = -1;

function render() {
  const person = currentPerson();
  if (!person) return;

  $("#header-title-text").textContent = `Camino a los ${fmt(person.goal)} puntos`;
  $("#header-date").textContent = fmtDate(state.today);
  $("#reward-text").textContent = person.reward || "—";

  const stats = computeStats(person);

  // Count-up del total
  const totalEl = $("#progress-total");
  const prevTotal = state.lastTotals[person.key] || 0;
  animateNumber(totalEl, prevTotal, stats.total, 1000);
  state.lastTotals[person.key] = stats.total;

  $("#progress-goal").textContent = fmt(stats.goal);

  // Anillo
  const circumference = 2 * Math.PI * 70;
  const ringFg = $("#ring-fg");
  ringFg.style.strokeDasharray = circumference;
  ringFg.style.strokeDashoffset = circumference * (1 - stats.pct / 100);

  // Spark que viaja al final del arco
  const spark = $("#ring-spark");
  const angle = (stats.pct / 100) * 360 - 90;
  const rad = angle * Math.PI / 180;
  const ringRect = $(".ring-wrap");
  const radius = ringRect.offsetWidth / 2 - (ringRect.offsetWidth / 2) * (10 / 160) - 4;
  const cx = ringRect.offsetWidth / 2;
  spark.style.left = (cx + Math.cos(rad) * radius) + "px";
  spark.style.top = (cx + Math.sin(rad) * radius) + "px";
  spark.classList.remove("visible");
  void spark.offsetWidth;
  if (stats.pct > 0 && stats.pct < 100) spark.classList.add("visible");

  // Porcentaje con count-up
  const pctEl = $("#ring-percent");
  const prevPctVal = parseInt(pctEl.textContent) || 0;
  animateNumber(pctEl, prevPctVal, Math.round(stats.pct), 1200, (v) => String(v));
  $("#ring-sub").textContent = stats.total >= stats.goal ? "¡Meta alcanzada!" : "de la meta";

  const progressCard = $("#progress-card");
  const progressBarFill = $("#progress-bar-fill");
  progressBarFill.style.width = stats.pct + "%";
  progressBarFill.setAttribute("aria-valuenow", Math.round(stats.pct));

  // Hitos
  [25, 50, 75].forEach(ms => {
    const reached = stats.pct >= ms;
    const mEl = $(`.milestone[data-ms="${ms}"]`);
    const lEl = $(`.milestone-label[data-msl="${ms}"]`);
    if (mEl) mEl.classList.toggle("reached", reached);
    if (lEl) lEl.classList.toggle("reached", reached);
  });

  const remaining = Math.max(0, stats.goal - stats.total);
  const remEl = $("#remaining-points");
  animateNumber(remEl, parseInt(remEl.textContent.replace(/\D/g,'')) || remaining, remaining, 900);

  $("#level-emoji").textContent = stats.level.emoji;
  $("#level-name").textContent = stats.level.name;

  // Estado dorado + confeti
  if (stats.total >= stats.goal) {
    progressCard.classList.add("gold");
    addSparkles(progressCard);
    if (prevPct !== -1 && prevPct < 100) triggerConfetti();
  } else {
    progressCard.classList.remove("gold");
    removeSparkles(progressCard);
  }
  prevPct = stats.pct;

  renderStats(stats);
  renderCategories(person);
  renderCategorySummary(stats);
  renderHistory(person);
  renderCharts(stats);
}

function addSparkles(container) {
  if (container.querySelector(".sparkle")) return;
  if (window.matchMedia("(prefers-reduced-motion: reduce)").matches) return;
  const positions = [
    { top: "12%", left: "8%", delay: "0s" },
    { top: "20%", right: "10%", delay: "0.8s" },
    { bottom: "18%", left: "14%", delay: "1.6s" },
    { bottom: "25%", right: "16%", delay: "0.4s" },
    { top: "50%", left: "4%", delay: "1.2s" }
  ];
  positions.forEach((pos, i) => {
    const s = document.createElement("span");
    s.className = "sparkle";
    s.textContent = i % 2 === 0 ? "✨" : "⭐";
    s.style.animationDelay = pos.delay;
    Object.assign(s.style, pos);
    container.appendChild(s);
  });
}

    }
  };

  const lineCtx = $("#chart-line").getContext("2d");
  const gradient = lineCtx.createLinearGradient(0, 0, 0, 240);
  gradient.addColorStop(0, colors.forest + "66");
  gradient.addColorStop(1, colors.forest + "00");
  chartLine = new Chart(lineCtx, {
    type: "line",
    data: {
      labels,
      datasets: [{
        label: "Acumulado",
        data: stats.series.accum,
        borderColor: colors.forest,
        backgroundColor: gradient,
        borderWidth: 3,
        fill: true,
        tension: 0.4,
        pointRadius: 0,
        pointHoverRadius: 6,
        pointHoverBackgroundColor: colors.forest,
        pointHoverBorderColor: "#fff",
        pointHoverBorderWidth: 2,
        pointHitRadius: 12
      }]
    },
    options: commonOpts
  });

  const barColors = stats.series.daily.map(v => v >= 0 ? colors.pos : colors.neg);
  chartBars = new Chart($("#chart-bars").getContext("2d"), {
    type: "bar",
    data: {
      labels,
      datasets: [{
        label: "Puntos",
        data: stats.series.daily,
        backgroundColor: barColors,
        hoverBackgroundColor: barColors.map(c => c),
        borderRadius: 5,
        borderSkipped: false,
        maxBarThickness: 20
      }]
    },
    options: {
      ...commonOpts,
      animation: reduceMotion ? false : {
        duration: 1200,
        easing: "easeOutQuart",
        delay: (ctx) => ctx.type === "data" ? ctx.dataIndex * 25 : 0
      }
    }
  });
}

/* ==========================================================================
   CONFETI MEJORADO
   ========================================================================== */
function triggerConfetti() {
  if (window.matchMedia("(prefers-reduced-motion: reduce)").matches) return;
  const canvas = $("#confetti");
  const ctx = canvas.getContext("2d");
  canvas.width = window.innerWidth;
  canvas.height = window.innerHeight;
  const particles = [];
  const colors = ["#c9a24f", "#e0c07a", "#1f5d3d", "#2d7a52", "#5bbd8e", "#fff5d1", "#f0d090"];
  const shapes = ["rect", "circle", "star"];

  for (let i = 0; i < 180; i++) {
    const angle = Math.random() * Math.PI * 2;
    const speed = Math.random() * 14 + 6;
    particles.push({
      x: canvas.width / 2,
      y: canvas.height * 0.35,
      vx: Math.cos(angle) * speed,
      vy: Math.sin(angle) * speed - 6,
      g: 0.32,
      size: Math.random() * 8 + 4,
      color: colors[Math.floor(Math.random() * colors.length)],
      shape: shapes[Math.floor(Math.random() * shapes.length)],
      rot: Math.random() * Math.PI * 2,
      vr: (Math.random() - 0.5) * 0.35,
      drag: 0.985
    });
  }

  const start = performance.now();
  function drawStar(x, y, r, rot) {
    ctx.save();
    ctx.translate(x, y);
    ctx.rotate(rot);
    ctx.beginPath();
    for (let i = 0; i < 5; i++) {
      const a = (i * 4 * Math.PI) / 5 - Math.PI / 2;
      const px = Math.cos(a) * r;
      const py = Math.sin(a) * r;
      if (i === 0) ctx.moveTo(px, py);
      else ctx.lineTo(px, py);
    }
    ctx.closePath();
    ctx.fill();
    ctx.restore();
  }

  function step(now) {
    const elapsed = now - start;
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    let alive = false;
    particles.forEach(p => {
      p.vx *= p.drag;
      p.vy = p.vy * p.drag + p.g;
      p.x += p.vx;
      p.y += p.vy;
      p.rot += p.vr;
      const alpha = Math.max(0, 1 - elapsed / 4200);
      if (alpha > 0 && p.y < canvas.height + 40) {
        alive = true;
        ctx.globalAlpha = alpha;
        ctx.fillStyle = p.color;
        if (p.shape === "rect") {
          ctx.save();
          ctx.translate(p.x, p.y);
          ctx.rotate(p.rot);
          ctx.fillRect(-p.size / 2, -p.size / 4, p.size, p.size / 2);
          ctx.restore();
        } else if (p.shape === "circle") {
          ctx.beginPath();
          ctx.arc(p.x, p.y, p.size / 2, 0, Math.PI * 2);
          ctx.fill();
        } else {
          drawStar(p.x, p.y, p.size / 1.6, p.rot);
        }
      }
    });
    ctx.globalAlpha = 1;
    if (alive && elapsed < 5000) requestAnimationFrame(step);
    else ctx.clearRect(0, 0, canvas.width, canvas.height);
  }
  requestAnimationFrame(step);
}

/* ==========================================================================
   SELECTOR DE PERSONA
   ========================================================================== */
function renderPersonSelector() {
  const container = $(".person-selector");
  const pill = $("#selector-pill");
  $$(".person-btn", container).forEach(b => b.remove());

  state.persons.forEach(p => {
    const btn = document.createElement("button");
    btn.type = "button";
    btn.className = "person-btn";
    btn.setAttribute("role", "tab");
    btn.setAttribute("aria-selected", p.key === state.selectedKey ? "true" : "false");
    btn.setAttribute("data-key", p.key);
    btn.innerHTML = `
      <span class="avatar">${esc(p.name.charAt(0).toUpperCase())}</span>
      <span class="person-name">${esc(p.name)}</span>
    `;
    btn.addEventListener("click", () => {
      if (state.selectedKey === p.key) return;
      state.selectedKey = p.key;
      prevPct = -1;
      renderPersonSelector();
      render();
    });
    container.appendChild(btn);
  });

  requestAnimationFrame(() => {
    const active = container.querySelector('.person-btn[aria-selected="true"]');
    if (active) {
      pill.style.width = active.offsetWidth + "px";
      pill.style.transform = `translateX(${active.offsetLeft - 4}px)`;
    }
  });
}

/* ==========================================================================
   REVEAL ON SCROLL
   ========================================================================== */
function setupReveal() {
  const els = $$(".reveal");
  if (!("IntersectionObserver" in window)) {
    els.forEach(e => e.classList.add("revealed"));
    return;
  }
  const io = new IntersectionObserver((entries) => {
    entries.forEach(entry => {
      if (entry.isIntersecting) {
        entry.target.classList.add("revealed");
        io.unobserve(entry.target);
      }
    });
  }, { threshold: 0.08, rootMargin: "0px 0px -40px 0px" });
  els.forEach(e => io.observe(e));
}

/* ==========================================================================
   TILT 3D sutil en escritorio
   ========================================================================== */
function setupTilt() {
  if (window.matchMedia("(hover: none)").matches) return;
  if (window.matchMedia("(prefers-reduced-motion: reduce)").matches) return;
  $$(".card").forEach(card => {
    card.addEventListener("mousemove", (e) => {
      const rect = card.getBoundingClientRect();
      const x = (e.clientX - rect.left) / rect.width - 0.5;
      const y = (e.clientY - rect.top) / rect.height - 0.5;
      card.style.transform = `perspective(1000px) rotateX(${-y * 2.2}deg) rotateY(${x * 2.2}deg) translateY(-2px)`;
    });
    card.addEventListener("mouseleave", () => {
      card.style.transform = "";
    });
  });
}

async function init() {
  const header = $("#header");
  const onScroll = () => {
    if (window.scrollY > 10) header.classList.add("scrolled");
    else header.classList.remove("scrolled");
  };
  window.addEventListener("scroll", onScroll, { passive: true });
  onScroll();

  $("#reg-date").value = state.today;
  $("#btn-manual").addEventListener("click", openManualModal);

  let resizeT;
  window.addEventListener("resize", () => {
    clearTimeout(resizeT);
    resizeT = setTimeout(() => {
      if (chartLine) chartLine.resize();
      if (chartBars) chartBars.resize();
      const cv = $("#confetti");
      if (cv) { cv.width = window.innerWidth; cv.height = window.innerHeight; }
      // Recalcular posición del spark
      if (state.loaded) {
        const person = currentPerson();
        if (person) {
          const stats = computeStats(person);
          const spark = $("#ring-spark");
          const angle = (stats.pct / 100) * 360 - 90;
          const rad = angle * Math.PI / 180;
          const ringRect = $(".ring-wrap");
          const radius = ringRect.offsetWidth / 2 - 14;
          const cx = ringRect.offsetWidth / 2;
          spark.style.left = (cx + Math.cos(rad) * radius) + "px";
          spark.style.top = (cx + Math.sin(rad) * radius) + "px";
        }
      }
    }, 150);
  });

  try {
    const data = await apiGetAll();
    state.today = data.today || todayISO();
    state.persons = data.persons || [];
    state.selectedKey = state.persons[0]?.key || null;
    state.loaded = true;

    $("#initial-loading").classList.add("hidden");
    $("#dashboard").classList.remove("hidden");

    renderPersonSelector();
    render();
    setupReveal();
    setupTilt();

    if (MOCK.enabled) {
      setTimeout(() => toast("Modo demo activo · los cambios no se guardan", "info", 4500), 800);
    }
  } catch (e) {
    $("#initial-loading").innerHTML = `
      <div class="empty" style="padding: 24px;">
        <div class="emoji">⚠️</div>
        <h3>Error al cargar</h3>
        <p>${esc(e.message || "No se pudieron obtener los datos.")}</p>
      </div>
    `;
    toast(e.message || "Error al cargar", "error", 5000);
  }
}
