<script>
const $ = id => document.getElementById(id);
const nf = n => Number(n).toLocaleString('es-CO');
const sg = n => (n > 0 ? '+' : '') + nf(n);
const esc = s => String(s == null ? '' : s).replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
const ymd = d => d.getFullYear() + '-' + String(d.getMonth() + 1).padStart(2, '0') + '-' + String(d.getDate()).padStart(2, '0');
const pd = s => new Date(s + 'T12:00:00');
const show = s => pd(s).toLocaleDateString('es-CO', { day: '2-digit', month: 'short', year: 'numeric' });

let S = { persons: [], today: '', cur: 0 }, busy = false, cA, cD;

google.script.run.withSuccessHandler(init).withFailureHandler(e => {
  $('loader').innerHTML = '<div>⚠️ ' + esc(e.message || e) + '</div>';
}).getAll();

function init(d) {
  S.persons = d.persons; S.today = d.today;
  $('date').value = d.today;
  $('loader').style.display = 'none';
  if (!S.persons.length) {
    document.body.innerHTML = '<div style="padding:60px;text-align:center">Aún no hay hojas. En tu Google Sheet ejecuta <b>Puntos › Configuración inicial</b>.</div>';
    return;
  }
  $('app').style.display = 'block';
  render();
}

const P = () => S.persons[S.cur];

function icon(n) {
  n = n.toLowerCase();
  if (n.includes('instal')) return '📱';
  if (n.includes('redes')) return '📵';
  if (n.includes('sexual')) return '🤐';
  if (n.includes('chiste mal') || n.includes('chistes mal')) return '🙄';
  if (n.includes('chiste')) return '😄';
  if (n.includes('ortogr')) return '✏️';
  if (n.includes('novia')) return '💘';
  if (n.includes('entrar')) return '🏁';
  if (n.includes('gym')) return '🏋️';
  if (n.includes('ausencia')) return '🚫';
  return '⭐';
}

function level(p) {
  return p >= 1 ? '🏆 Meta alcanzada' : p >= .75 ? '🏁 Recta final' : p >= .5 ? '⚡ A mitad de camino' : p >= .25 ? '🔥 En marcha' : '🌱 Comenzando';
}

function stats(p) {
  const by = {}; let total = 0, gained = 0, lost = 0;
  p.records.forEach(r => { total += r.points; r.points > 0 ? gained += r.points : lost += r.points; by[r.date] = (by[r.date] || 0) + r.points; });
  let best = null; Object.keys(by).forEach(k => { if (!best || by[k] > best.v) best = { d: k, v: by[k] }; });
  let streak = 0, d = pd(S.today); if (!(by[ymd(d)] > 0)) d.setDate(d.getDate() - 1);
  while (by[ymd(d)] > 0) { streak++; d.setDate(d.getDate() - 1); }
  return { by, total, gained, lost, best, streak, today: by[S.today] || 0,
    todayN: p.records.filter(r => r.date === S.today).length,
    pct: p.goal > 0 ? Math.max(0, Math.min(1, total / p.goal)) : 0 };
}

function render() {
  const p = P(), s = stats(p);
  $('title').textContent = 'Camino a los ' + nf(p.goal) + ' puntos';
  $('today').textContent = show(S.today);
  $('reward').textContent = 'Recompensa al llegar a la meta: ' + p.reward;
  $('seg').innerHTML = S.persons.map((x, i) =>
    '<button class="' + (i === S.cur ? 'on' : '') + '" onclick="pick(' + i + ')"><span class="av">' + esc(x.name.charAt(0)) + '</span>' + esc(x.name) + '</button>').join('');

  $('total').textContent = nf(s.total); $('goal').textContent = nf(p.goal);
  $('pct').textContent = Math.round(s.pct * 100) + '%';
  $('ringC').style.strokeDashoffset = 402.1 * (1 - s.pct);
  $('fill').style.width = s.pct * 100 + '%';
  $('level').textContent = level(s.pct);
  $('left').innerHTML = s.total >= p.goal ? '🎉 ¡Meta cumplida! Reclama la recompensa.'
    : 'Faltan <strong>' + nf(p.goal - s.total) + '</strong> puntos para la recompensa.';

  $('stats').innerHTML = [
    ['Puntos ganados', '<span class="pos">' + sg(s.gained) + '</span>', 'suman al total'],
    ['Puntos perdidos', '<span class="neg">' + sg(s.lost) + '</span>', 'restan del total'],
    ['Hoy', '<span class="' + (s.today > 0 ? 'pos' : s.today < 0 ? 'neg' : '') + '">' + sg(s.today) + '</span>', s.todayN + ' registros hoy'],
    ['Racha', s.streak + (s.streak === 1 ? ' día' : ' días'), 'días seguidos sumando'],
    ['Mejor día', s.best ? sg(s.best.v) : '—', s.best ? show(s.best.d) : 'sin registros'],
    ['Registros', nf(p.records.length), 'en total']
  ].map(x => '<div class="card stat"><div class="lbl">' + x[0] + '</div><div class="v">' + x[1] + '</div><div class="s">' + x[2] + '</div></div>').join('');

  $('chips').innerHTML = p.categories.map((c, i) =>
    '<button class="chip ' + (c.points === null ? '' : c.points > 0 ? 'p' : 'n2') + '" onclick="tap(' + i + ')"><span class="ic">' + icon(c.name) +
    '</span><span><div class="n">' + esc(c.name) + '</div><div class="pt">' + (c.points === null ? 'a criterio' : sg(c.points)) + '</div></span></button>').join('');

  const cs = {}; p.records.forEach(r => { const o = cs[r.category] || (cs[r.category] = { n: 0, p: 0 }); o.n++; o.p += r.points; });
  const names = p.categories.map(c => c.name); Object.keys(cs).forEach(k => { if (names.indexOf(k) < 0) names.push(k); });
  const mx = Math.max(1, ...names.map(k => Math.abs((cs[k] || { p: 0 }).p)));
  $('cats').innerHTML = '<div class="row h"><div>Categoría</div><div style="text-align:center">Reg.</div><div></div><div style="text-align:right">Puntos</div></div>' +
    names.map(k => { const o = cs[k] || { n: 0, p: 0 };
      return '<div class="row"><div>' + icon(k) + ' ' + esc(k) + '</div><div class="cnt">' + o.n + '</div><div class="t"><i style="width:' + Math.abs(o.p) / mx * 100 + '%;background:' + (o.p < 0 ? 'var(--neg)' : 'var(--green2)') + '"></i></div><div class="num ' + (o.p > 0 ? 'pos' : o.p < 0 ? 'neg' : '') + '">' + sg(o.p) + '</div></div>'; }).join('');

  const hs = p.records.slice().sort((a, b) => a.date < b.date ? 1 : a.date > b.date ? -1 : b.id - a.id).slice(0, 40);
  $('hist').innerHTML = hs.length ? hs.map(r =>
    '<div class="h-it"><div class="ic">' + icon(r.category) + '</div><div class="m"><b>' + esc(r.category) + '</b><span>' + show(r.date) + (r.reason ? ' · ' + esc(r.reason) : '') +
    '</span></div><div class="pts ' + (r.points > 0 ? 'pos' : r.points < 0 ? 'neg' : '') + '">' + sg(r.points) + '</div><button class="x" title="Eliminar" onclick="askDel(' + r.id + ')">✕</button></div>').join('')
    : '<div class="empty">Sin registros todavía.</div>';

  charts(p);
}

function charts(p) {
  const by = {}, labels = [], acc = [], day = [];
  p.records.forEach(r => by[r.date] = (by[r.date] || 0) + r.points);
  const start = pd(S.today); start.setDate(start.getDate() - 29);
  let run = 0; const s0 = ymd(start);
  p.records.forEach(r => { if (r.date < s0) run += r.points; });
  for (let i = 0; i < 30; i++) {
    const d = new Date(start); d.setDate(start.getDate() + i);
    const k = ymd(d), v = by[k] || 0; run += v;
    labels.push(d.toLocaleDateString('es-CO', { day: '2-digit', month: 'short' })); acc.push(run); day.push(v);
  }
  const base = { responsive: true, maintainAspectRatio: false, plugins: { legend: { display: false } },
    scales: { x: { grid: { display: false }, ticks: { font: { size: 10 }, maxTicksLimit: 8, color: '#7b857f' } },
              y: { grid: { color: '#f0ede2' }, ticks: { font: { size: 10 }, color: '#7b857f' } } } };
  if (cA) cA.destroy(); if (cD) cD.destroy();
  cA = new Chart($('cA'), { type: 'line', data: { labels, datasets: [{ data: acc, borderColor: '#1f5d3d', backgroundColor: 'rgba(31,93,61,.1)', fill: true, tension: .35, pointRadius: 0, pointHoverRadius: 5, borderWidth: 2.5 }] },
    options: Object.assign({}, base, { interaction: { mode: 'index', intersect: false }, plugins: { legend: { display: false }, title: { display: true, text: 'Total acumulado', font: { size: 13, weight: '600' }, color: '#1d2a24' } } }) });
  cD = new Chart($('cD'), { type: 'bar', data: { labels, datasets: [{ data: day, backgroundColor: day.map(v => v >= 0 ? 'rgba(47,138,92,.75)' : 'rgba(192,57,43,.75)'), borderRadius: 5 }] },
    options: Object.assign({}, base, { plugins: { legend: { display: false }, title: { display: true, text: 'Puntos por día', font: { size: 13, weight: '600' }, color: '#1d2a24' } } }) });
}

function pick(i) { S.cur = i; render(); }

function setBusy(v) {
  busy = v; document.body.classList.toggle('busy', v);
  document.querySelectorAll('.chip,.btn,.x').forEach(b => b.disabled = v);
}
function toast(m, err) {
  const t = $('toast'); t.textContent = m; t.className = 'toast on' + (err ? ' err' : '');
  clearTimeout(toast.t); toast.t = setTimeout(() => t.className = 'toast', 2800);
}

function send(fnName, args, okMsg) {
  if (busy) return;
  setBusy(true);
  const runner = google.script.run
    .withSuccessHandler(res => { S.persons[S.cur] = res; setBusy(false); render(); closeModal(); toast(okMsg); })
    .withFailureHandler(e => { setBusy(false); toast(e.message || String(e), true); });
  runner[fnName].apply(runner, args);
}

function tap(i) {
  const c = P().categories[i];
  if (c.points === null) return openManual(c.name);
  send('addRecord', [P().key, { date: $('date').value, category: c.name, points: c.points, reason: $('note').value }], '✔ ' + c.name + ' (' + sg(c.points) + ')');
  $('note').value = '';
}

function openModal(html) { $('box').innerHTML = html; $('modal').classList.add('on'); }
function closeModal() { $('modal').classList.remove('on'); }
$('modal').addEventListener('click', e => { if (e.target === $('modal')) closeModal(); });

function openManual(cat) {
  const p = P();
  openModal('<h3>Registro manual</h3>' +
    '<div class="f"><label>Categoría</label><select id="mC">' + p.categories.map(c => '<option value="' + esc(c.name) + '"' + (c.name === cat ? ' selected' : '') + '>' + esc(c.name) + (c.points === null ? '' : ' (' + sg(c.points) + ')') + '</option>').join('') + '</select></div>' +
    '<div class="f"><label>Puntos (vacío = valor por defecto)</label><input type="number" id="mP" placeholder="Ej: 25 o -10"></div>' +
    '<div class="f"><label>Fecha</label><input type="date" id="mD" value="' + esc($('date').value) + '"></div>' +
    '<div class="f"><label>Motivo (opcional)</label><input type="text" id="mR" value="' + esc($('note').value) + '"></div>' +
    '<div class="acts"><button class="btn ghost" onclick="closeModal()">Cancelar</button><button class="btn" onclick="saveManual()">Registrar</button></div>');
  if (cat) setTimeout(() => $('mP').focus(), 50);
}

}
